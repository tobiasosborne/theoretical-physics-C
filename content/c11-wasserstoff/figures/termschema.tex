% c11.fig.termschema — term scheme of the relative motion with the Lyman and Balmer transitions.
% Reworked from latex/figures/fig_vl16_termschema.tex, panel (a).
%
% Generating formula (DERIVATION.md Results 2.2 and 8.2, Z = 1):
%     E_n / Ry_mu = -1/n^2,   Ry_mu = (mu/m_e) Ry  (the Rydberg energy WITH THE REDUCED MASS).
% The vertical axis is E/Ry_mu, so no numerical constant enters the picture.
% Levels n = 1..16 are drawn, each at its computed height (pgfplots evaluates -1/n^2).
% Arrows: Lyman n -> 1 for n = 2, 3, 4 (solid); Balmer n -> 2 for n = 3, 4, 5 (dashed).
% The length of an arrow is the energy difference E_n - E_n', to scale.
% Greyscale: the two series differ by dash pattern and are labelled directly.
\begin{tikzpicture}[>={Stealth[length=5pt,width=3.6pt]}]
  \pgfplotsset{c11/.style={
    clip=false, grid=none,
    axis line style={draw=tpc-plate-ink, line width=\tpcstrokethin},
    tick style={draw=tpc-plate-ink, line width=\tpcstrokehair},
    tick label style={font=\tpcfontlabel, text=tpc-plate-ink},
    label style={font=\tpcfontlabel, text=tpc-plate-ink}}}
  \begin{axis}[c11, width=8.6cm, height=7.4cm,
      axis y line=left, axis x line=none,
      xmin=0, xmax=1.25, ymin=-1.075, ymax=0.09,
      ytick={0,-0.1111111,-0.25,-1},
      yticklabels={$0$,$-\tfrac19$,$-\tfrac14$,$-1$},
      ylabel={$E/\mathrm{Ry}_{\mu}$},
      every axis y label/.style={at={(ticklabel* cs:1.0)}, anchor=south, font=\tpcfontlabel}]
    \pgfplotsinvokeforeach{1,...,16}{
      \draw[tpc leader, draw=tpc-plate-muted] (axis cs:0.03,{-1/(#1*#1)}) -- (axis cs:0.93,{-1/(#1*#1)});}
    \draw[tpc axis, line width=\tpcstrokeregular] (axis cs:0.03,0) -- (axis cs:0.93,0);
    % Lyman n -> 1
    \draw[tpc q1, ->] (axis cs:0.14,-0.25)  -- (axis cs:0.14,-1);
    \draw[tpc q1, ->] (axis cs:0.22,{-1/9})  -- (axis cs:0.22,-1);
    \draw[tpc q1, ->] (axis cs:0.30,{-1/16}) -- (axis cs:0.30,-1);
    % Balmer n -> 2
    \draw[tpc q4, dash pattern=on 3.5pt off 1.5pt, ->] (axis cs:0.58,{-1/9})  -- (axis cs:0.58,-0.25);
    \draw[tpc q4, dash pattern=on 3.5pt off 1.5pt, ->] (axis cs:0.66,{-1/16}) -- (axis cs:0.66,-0.25);
    \draw[tpc q4, dash pattern=on 3.5pt off 1.5pt, ->] (axis cs:0.74,{-1/25}) -- (axis cs:0.74,-0.25);
    \node[tpc label, anchor=west, align=left] at (axis cs:0.34,-0.63)
      {Lyman\\ $n \to 1$};
    \node[tpc label, anchor=north, align=center] at (axis cs:0.66,-0.275)
      {Balmer\\ $n \to 2$};
    \node[tpc label, anchor=west] at (axis cs:0.95,-1) {$n = 1$};
    \node[tpc label, anchor=west] at (axis cs:0.95,-0.25) {$n = 2$};
    \node[tpc label, anchor=west] at (axis cs:0.95,-0.125) {$n = 3$};
    \node[tpc label, anchor=west] at (axis cs:0.95,0.02) {$n \to \infty$};
    \node[tpc label, anchor=south] at (axis cs:0.48,0.01)
      {\tpcifenglish{$E = 0$: ionisation threshold}{$E = 0$: Ionisationsschwelle}};
  \end{axis}
\end{tikzpicture}
