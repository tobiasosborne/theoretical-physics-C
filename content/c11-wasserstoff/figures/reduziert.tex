% c11.fig.reduziert — the reduced mass as a function of the mass ratio.
%
% Generating formula (registry definition D62): mu = m_e m_p/(m_e + m_p), hence
%     mu/m_e = 1/(1 + m_e/m_p) = t/(1 + t),   t = m_p/m_e.
% Marked: t = 1 (two equal masses, mu/m_e = 1/2) and hydrogen, t = 1/q with
% q = m_e/m_p = 5.446170214889e-4 (CODATA 2022, allascii.txt line 117), where
% mu/m_e = 1/(1 + q) = 0.999455679424762 (DERIVATION.md Result 8.2).
% The horizontal axis is logarithmic.
% Greyscale: one curve, two labelled points.
\begin{tikzpicture}
  \pgfplotsset{c11/.style={
    axis lines=left, clip=false, grid=none,
    axis line style={draw=tpc-plate-ink, line width=\tpcstrokethin},
    tick style={draw=tpc-plate-ink, line width=\tpcstrokehair},
    tick label style={font=\tpcfontlabel, text=tpc-plate-ink},
    label style={font=\tpcfontlabel, text=tpc-plate-ink},
    every axis x label/.style={at={(ticklabel* cs:1.0)}, anchor=west, font=\tpcfontlabel},
    every axis y label/.style={at={(ticklabel* cs:1.0)}, anchor=south, font=\tpcfontlabel}}}
  \begin{axis}[c11, width=9.4cm, height=4.6cm,
      xmode=log, xmin=0.1, xmax=20000, ymin=0, ymax=1.12,
      xtick={0.1,1,10,100,1000,10000},
      xticklabels={$\tfrac{1}{10}$,$1$,$10$,$10^2$,$10^3$,$10^4$},
      ytick={0,0.5,1}, yticklabels={$0$,$\tfrac12$,$1$},
      xlabel={$m_{\mathrm{p}}/m_{\mathrm{e}}$}, ylabel={$\mu/m_{\mathrm{e}}$}]
    \draw[tpc grid] (axis cs:0.1,1) -- (axis cs:20000,1);
    \addplot[tpc dynamics, domain=0.1:20000, samples=200] {x/(1+x)};
    \fill[tpc-plate-ink] (axis cs:1,0.5) circle (1.6pt);
    \fill[tpc-plate-ink] (axis cs:{1/5.446170214889e-4},{1/(1+5.446170214889e-4)}) circle (1.6pt);
    \node[tpc label, anchor=north west] at (axis cs:1.15,0.5) {\tpcifenglish{equal masses}{gleiche Massen}};
    \node[tpc label, anchor=north] at (axis cs:{1/5.446170214889e-4},0.93) {\tpcifenglish{hydrogen}{Wasserstoff}};
  \end{axis}
\end{tikzpicture}
