% c11.fig.laguerre — the generalised Laguerre polynomials L_k^(1), k = 0, 1, 2, 3.
%
% Generating formula (registry definition D68; DERIVATION.md Result 3.1):
%     L_k^(alpha)(x) = sum_{i=0}^{k} (-1)^i binom(k + alpha, k - i) x^i / i!,
% for alpha = 1 (the polynomials of the s states, R_n0 ~ L_{n-1}^(1)):
%     L_0 = 1,
%     L_1 = 2 - x,
%     L_2 = 3 - 3x + x^2/2,
%     L_3 = 4 - 6x + 2x^2 - x^3/6.
% L_k^(1)(0) = k + 1.  Greyscale: four dash patterns, labelled directly.
\begin{tikzpicture}
  \pgfplotsset{c11/.style={
    axis lines=left, clip=false, grid=none,
    axis line style={draw=tpc-plate-ink, line width=\tpcstrokethin},
    tick style={draw=tpc-plate-ink, line width=\tpcstrokehair},
    tick label style={font=\tpcfontlabel, text=tpc-plate-ink},
    label style={font=\tpcfontlabel, text=tpc-plate-ink},
    every axis x label/.style={at={(ticklabel* cs:1.0)}, anchor=west, font=\tpcfontlabel},
    every axis y label/.style={at={(ticklabel* cs:1.0)}, anchor=south, font=\tpcfontlabel}}}
  \begin{axis}[c11, width=9.0cm, height=5.2cm,
      xmin=0, xmax=8.6, ymin=-3.6, ymax=4.6,
      xtick={0,2,4,6,8}, ytick={-2,0,2,4},
      xlabel={$x$}, ylabel={$L_k^{(1)}(x)$}]
    \draw[tpc grid] (axis cs:0,0) -- (axis cs:8.6,0);
    \addplot[tpc q1, domain=0:8.6, samples=2] {1};
    \addplot[tpc q3, dash pattern=on 5pt off 2pt, domain=0:5.6, samples=2] {2-x};
    \addplot[tpc q4, dash pattern=on 5pt off 1.5pt on 1.2pt off 1.5pt, domain=0:6.46, samples=80] {3-3*x+x*x/2};
    \addplot[tpc q5, dash pattern=on 1.5pt off 1.5pt, domain=0:8.3, samples=120] {4-6*x+2*x*x-x^3/6};
    \node[tpc label, anchor=south] at (axis cs:7.9,1.1) {$k = 0$};
    \node[tpc label] at (axis cs:3.2,-2.7) {$k = 1$};
    \node[tpc label, anchor=west] at (axis cs:6.6,4.25) {$k = 2$};
    \node[tpc label, anchor=south east] at (axis cs:4.2,2.6) {$k = 3$};
  \end{axis}
\end{tikzpicture}
