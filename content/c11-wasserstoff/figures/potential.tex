% c11.fig.potential — effective potential of the relative motion for l = 0 and l = 1.
%
% Generating formula (registry definition D71; DERIVATION.md Result 2.1, Z = 1):
%     V_l^eff(r) = hbar^2 l(l+1)/(2 mu r^2) - e^2/(4 pi eps_0 r).
% In units of the Bohr radius WITH THE REDUCED MASS, x = r/a_mu, and of the Rydberg energy
% WITH THE REDUCED MASS, Ry_mu = hbar^2/(2 mu a_mu^2) = e^2/(8 pi eps_0 a_mu):
%     V_l^eff / Ry_mu = l(l+1)/x^2 - 2/x.
% l = 1: minimum at x = 2 with value -1/2 (derivative -4/x^3 + 2/x^2 = 0).
% Energy levels of DERIVATION.md Result 2.2: E_n/Ry_mu = -1/n^2, drawn for n = 1, 2, 3.
% Greyscale: l = 0 solid, l = 1 dashed, levels thin.

    axis lines=left, clip=false, grid=none,

  \begin{axis}[c11, width=9.4cm, height=6.2cm,

      yticklabels={$-2$,$-1$,$-\tfrac14$,$0$,$1$},
