% c02.fig.zustandsraeume — the three state spaces of the chapter, each with a mixture.
%
% (a) classical bit: the segment between the pure states s_0 and s_1, drawn along the
%     coordinate p_1 (so s_0 is at 0 and s_1 at 1).  x: p_1 = 1/8, y: p_1 = 7/8,
%     z = 1/4 x + 3/4 y: p_1 = 1/4*1/8 + 3/4*7/8 = 22/32 = 11/16.
% (b) square bit: K_Q = [-1,1]^2.  x = (-3/4, 1/2), y = (3/4, -1/4),
%     z = 1/4 x + 3/4 y = (-3/16 + 9/16, 1/8 - 3/16) = (3/8, -1/16).
% (c) qubit: unit ball in R^3 in ORTHOGRAPHIC projection, so the outline is the unit
%     circle.  Viewing angles alpha = 25 deg (azimuth), beta = 15 deg (elevation):
%       e_x -> (-sin a, -sin b cos a), e_y -> (cos a, -sin b sin a), e_z -> (0, cos b).
%     Equator (r_z = 0): the half with cos(t - alpha) > 0 faces the viewer (solid).
%     x = (0, -3/4, 1/2), y = (0, 3/4, -1/4), z = 1/4 x + 3/4 y = (0, 3/8, -1/16).
% Greyscale: states are filled discs, the mixture is an open ring; pure states are
% squares in panels (a), (b); the boundary of the ball is the heavy circle.
\begin{tikzpicture}[x=1cm, y=1cm]
  % ---------- (a) bit ----------
  \begin{scope}[shift={(0,0)}, x=2.6cm]
    \draw[tpc axis] (0,0) -- (1,0);

    \fill[tpc-plate-ink] (0.978,-0.06) rectangle (1.022,0.06);
    \node[tpc label, below=3pt] at (0,0) {$s_0$};
    \node[tpc label, below=3pt] at (1,0) {$s_1$};

    \fill[tpc-state] (0.125,0) circle (2pt);
    \fill[tpc-state] (0.875,0) circle (2pt);

    \node[tpc label, above=3pt] at (0.875,0) {$y$};
    \node[tpc label, above=3pt] at (0.6875,0) {$z$};
    \node[tpc label] at (0.5,-1.75) {(a) \tpcifenglish{classical bit}{klassisches Bit}};
  \end{scope}

  \begin{scope}[shift={(5.2,0)}, x=1.15cm, y=1.15cm]

    \fill[tpc-state] (-0.75,0.5) circle (2pt);
    \fill[tpc-state] (0.75,-0.25) circle (2pt);

    \node[tpc label, above=3pt] at (-0.75,0.5) {$x$};
    \node[tpc label, below=3pt] at (0.75,-0.25) {$y$};
    \node[tpc label, above=3pt] at (0.375,-0.0625) {$z$};
    \node[tpc label] at (0,-1.52) {(b) \tpcifenglish{square bit}{Quadrat-Bit}};
