% c01.fig.indikator — a yes-no measurement and its expectation value.
%
% Outcome set X = {1,...,6}; event S = {2,4,6}; loaded die p = (1,2,2,2,2,3)/12
% (c01.eq.gezinkt).
%   (a) the indicator function chi_S (c01.eq.indikator-def): value 1 on S, 0 elsewhere;
%       scale 1 = 1.6 cm.
%   (b) the weight p(x) (outlined bars) and the products p(x) chi_S(x) (hatched): only the
%       bars over S remain; their heights add up to E[chi_S] = P[S] = (2+2+3)/12 = 7/12
%       (c01.eq.indikator).  Scale: 1 unit of probability = 9.6 cm.
% Greyscale: the measurement is drawn in the effect role (dashed outline), the weight in
% the state role (solid); the products are hatched.

    \foreach \x in {1,...,6} \node[tpc label, anchor=north] at (\x,-0.04) {$\x$};

    \node[tpc label, anchor=south] at (0.3,1.22) {$\chi_S(x)$};
    \node[tpc label, anchor=north] at (3.5,-0.36) {(a) $S = \{2,4,6\}$};
