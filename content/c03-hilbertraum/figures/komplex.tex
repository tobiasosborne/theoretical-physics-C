% c03.fig.komplex — a complex-valued function needs three dimensions.
%
% f(x) = exp(-x^2/2) exp(2ix), x in [-3, 3]:  (x, Re f, Im f) =
%   (x, exp(-x^2/2) cos 2x, exp(-x^2/2) sin 2x), drawn in an oblique projection
%   x -> (1.25 cm, 0), Re -> (0, 1.25 cm), Im -> (0.85 cm, 0.55 cm).
% Thin circles: the points at distance |f(x)| = exp(-x^2/2) from the x-axis in the planes
%   x = -1, 0, 1 (the curve passes through one point of each).
% Hairline spokes: from (x, 0, 0) to f(x) for x = -2, -1.75, ..., 2, as a depth cue.
% Dashed: Re f alone in the plane Im f = 0, the usual drawing.
% Greyscale: the curve in the state role (solid), Re f in the effect role (dashed),
% circles and spokes hairline.
\begin{tikzpicture}[x={(1.25cm,0cm)}, y={(0cm,1.25cm)}, z={(0.85cm,0.55cm)}]
  \foreach \c in {-1,0,1} {
    \draw[tpc grid] plot[domain=0:360, samples=73, variable=\t]
      (\c,{exp(-\c*\c/2)*cos(\t)},{exp(-\c*\c/2)*sin(\t)});
  }
  \draw[tpc axis, -{Stealth[length=5pt]}] (-3.3,0,0) -- (3.5,0,0);
  \draw[tpc axis, -{Stealth[length=5pt]}] (0,-1.2,0) -- (0,1.35,0);
  \draw[tpc axis, -{Stealth[length=5pt]}] (0,0,-1.25) -- (0,0,1.55);
  \node[tpc label, anchor=west] at (3.52,0,0) {$x$};
  \node[tpc label, anchor=south] at (0,1.37,0) {$\operatorname{Re} f$};
  \node[tpc label, anchor=south west] at (0,0,1.57) {$\operatorname{Im} f$};
  \foreach \s in {-2,-1.75,...,2.01} {
    \draw[tpc leader] (\s,0,0) -- (\s,{exp(-\s*\s/2)*cos(deg(2*\s))},{exp(-\s*\s/2)*sin(deg(2*\s))});
  }
  \draw[tpc effect] plot[domain=-3:3, samples=200] (\x,{exp(-\x*\x/2)*cos(deg(2*\x))},0);
  \draw[tpc state] plot[domain=-3:3, samples=300] (\x,{exp(-\x*\x/2)*cos(deg(2*\x))},{exp(-\x*\x/2)*sin(deg(2*\x))});
  \node[tpc label, anchor=south east] at (-1.05,0.55,0.6) {$f(x)$};
  \node[tpc label, anchor=north west] at (1.6,-0.85,0) {\tpcifenglish{dashed: $\operatorname{Re} f$ alone}{gestrichelt: nur $\operatorname{Re} f$}};
\end{tikzpicture}
