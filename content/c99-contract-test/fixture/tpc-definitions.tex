\tpcdeclaredef{T1}{kern}{Spur}{Trace}%
  {Die \emph{Spur} einer $2\times 2$-Matrix $X$ mit Einträgen $x_{jk}$ ist die
   Summe ihrer Diagonaleinträge, $\tr X = x_{11} + x_{22}$.}%
  {The \emph{trace} of a $2\times 2$ matrix $X$ with entries $x_{jk}$ is the
   sum of its diagonal entries, $\tr X = x_{11} + x_{22}$.}
\tpcdeclaredef{T2}{vertiefung}{Kommutator}{Commutator}%
  {Der \emph{Kommutator} zweier $2\times 2$-Matrizen $X$ und $Y$ ist die Matrix
   $\comm{X}{Y} = XY - YX$.}%
  {The \emph{commutator} of two $2\times 2$ matrices $X$ and $Y$ is the matrix
   $\comm{X}{Y} = XY - YX$.}
