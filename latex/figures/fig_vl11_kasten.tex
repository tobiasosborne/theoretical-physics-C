% fig_vl11_kasten.tex — Teilchen im Kasten: Eigenfunktionen auf ihren
% Energieniveaus, und die quadratische Aufspreizung des Spektrums.
%
% Reference figure for the notes: the other lecture figures follow the
% conventions established here.
%
%   * geometry and curves come from the closed-form solution derived in
%     the lecture, never from a freehand sketch:
%         phi_n(x) = sqrt(2/L) sin(n pi x / L),
%         E_n      = n^2 pi^2 hbar^2 / (2 m L^2),
%     so the node count, the symmetry and the level spacing are forced
%     to be right rather than drawn by eye.  The lecture calls the
%     eigenfunctions phi_n, and so does this plate;
%   * each eigenfunction is drawn about its own energy — the standard way
%     of showing that phi_n belongs to E_n;
%   * colour encodes the quantum number n (the ordered ramp of
%     `tpclevels`), and nothing else.  Entries 1, 2, 4, 5 (navy, teal,
%     orange, red); entry 3, munsell, is skipped on purpose — beside
%     cgblue it is the same teal to the eye, so it would break the very
%     code it is supposed to carry.  The energy levels themselves are
%     neutral grey rules: they are reference lines, not data;
%   * the classically forbidden region is hatched, not filled, so it
%     survives greyscale printing.  The hatching is a band beside each
%     wall rather than a full-bleed field, which leaves the level
%     labels standing on clean paper instead of on the hatch;
%   * the walls rise from the floor of the well and stop there, so the
%     x-axis marks 0 and L are not struck through by them;
%   * no legend where a direct label will do: in (b) each curve is named
%     where it runs.
%
% Panel (a) the first four eigenfunctions in the well.
% Panel (b) the spectrum E_n/E_1, with the oscillator's evenly spaced
%           ladder alongside it — the contrast between a quadratic and a
%           linear spectrum is the point students should carry away.
%           Both spectra are normalised to their own E_1, so both pass
%           through (1,1) and only the shape is being compared.  Today's
%           subject is the box, so it is the coloured one; the oscillator
%           is the grey reference it is measured against.

\begin{center}
\begin{tikzpicture}

% ─────────────────────────────────────────────────────────────
%  (a)  Eigenfunctions drawn about their levels
% ─────────────────────────────────────────────────────────────
\begin{axis}[
    name=wells,
    tpcplot,
    width=8.2cm, height=6.8cm,
    axis lines=none,
    xmin=-0.50, xmax=1.50,
    ymin=-2.2,  ymax=20.4,
    clip=false,
    title={(a) \tpcifenglish{$\phi_n$ on their levels}%
                       {$\phi_n$ auf ihren Niveaus}},
    domain=0:1, samples=201,
]

  % Classically forbidden regions, x<0 and x>L, where V = infinity.
  % A band beside each wall: enough to say "forbidden", not so much that
  % the labels have to sit on top of it.
  \addplot[forbidden, draw=none, domain=-0.26:0, samples=2] {18.6} \closedcycle;
  \addplot[forbidden, draw=none, domain=1:1.26,  samples=2] {18.6} \closedcycle;

  % The well itself: walls from the floor upwards, floor between them.
  \draw[potential] (axis cs:0,0) -- (axis cs:0,18.6);
  \draw[potential] (axis cs:1,0) -- (axis cs:1,18.6);
  \draw[potential] (axis cs:0,0) -- (axis cs:1,0);

  % Energy levels E_n = n^2 E_1: neutral reference rules
  \addplot[levellinegrey, domain=0:1, samples=2] {1};
  \addplot[levellinegrey, domain=0:1, samples=2] {4};
  \addplot[levellinegrey, domain=0:1, samples=2] {9};
  \addplot[levellinegrey, domain=0:1, samples=2] {16};

  % Eigenfunctions about those levels.  Amplitude 1.3 is the largest that
  % keeps phi_2 clear of E_1, the tightest gap (E_2 - E_1 = 3 E_1).
  \addplot[wavefn, spacecadet]   {1  + 1.3*sin(deg(1*pi*x))};
  \addplot[wavefn, cgblue]       {4  + 1.3*sin(deg(2*pi*x))};
  \addplot[wavefn, birrenorange] {9  + 1.3*sin(deg(3*pi*x))};
  \addplot[wavefn, birrenred]    {16 + 1.3*sin(deg(4*pi*x))};

  % Level labels.  Written out rather than looped: \foreach defers
  % expansion inside a pgfplots axis.
  \node[lbl, anchor=east] at (axis cs:-0.30,1)  {$E_{1}$};
  \node[lbl, anchor=east] at (axis cs:-0.30,4)  {$E_{2}$};
  \node[lbl, anchor=east] at (axis cs:-0.30,9)  {$E_{3}$};
  \node[lbl, anchor=east] at (axis cs:-0.30,16) {$E_{4}$};

  % Node counts: phi_n has n-1 interior nodes
  \node[lbl, text=birrengray, font=\scriptsize, anchor=west]
    at (axis cs:1.30,1)  {$n=1$};
  \node[lbl, text=birrengray, font=\scriptsize, anchor=west]
    at (axis cs:1.30,4)  {$n=2$};
  \node[lbl, text=birrengray, font=\scriptsize, anchor=west]
    at (axis cs:1.30,9)  {$n=3$};
  \node[lbl, text=birrengray, font=\scriptsize, anchor=west]
    at (axis cs:1.30,16) {$n=4$};

  % Axis furniture
  \node[lbl, anchor=north] at (axis cs:0,-0.35)    {$0$};
  \node[lbl, anchor=north] at (axis cs:1,-0.35)    {$L$};
  \node[lbl, anchor=north] at (axis cs:0.5,-0.95)  {$x$};
  \node[lbl, text=birrengray, font=\scriptsize, anchor=south]
    at (axis cs:-0.13,18.9) {$V=\infty$};
\end{axis}

% ─────────────────────────────────────────────────────────────
%  (b)  Spectrum: the n^2 ladder against the oscillator's linear one
% ─────────────────────────────────────────────────────────────
\begin{axis}[
    tpcplot,
    at={(wells.south east)}, anchor=south west, xshift=17mm,
    width=6.2cm, height=6.8cm,
    xlabel={$n$},
    ylabel={$E_n/E_1$},
    every axis y label/.style={at={(axis description cs:-0.16,0.5)},
                               rotate=90, anchor=south, font=\small},
    xmin=0.4, xmax=5.6, ymin=0, ymax=27,
    xtick={1,2,3,4,5}, ytick={0,5,10,15,20,25},
    grid=major,
    title={(b) \tpcifenglish{Spectrum}{Spektrum}},
]
  % Oscillator, normalised the same way: E_n/E_1 = (n + 1/2)/(3/2).
  % Grey and dashed: it is the reference this lecture is measured
  % against, not this lecture's own result.  Drawn first so the box
  % curve crosses over it.
  \addplot[birrengray, line width=0.9pt, densely dashed,
           domain=1:5, samples=2] {(2*x+1)/3};
  \addplot[birrengray, only marks, mark=o, mark size=1.9pt,
           domain=1:5, samples=5] {(2*x+1)/3};

  % Kasten: E_n / E_1 = n^2.  The smooth arc is the trend, the dots are
  % the levels — the spectrum exists only at integer n.
  \addplot[cgblue, line width=1.0pt, domain=1:5, samples=101] {x^2};
  \addplot[cgblue, only marks, mark=*, mark size=1.9pt,
           domain=1:5, samples=5] {x^2};

  % Direct labels instead of a legend box
  \node[lbl, text=cgblue, anchor=east] at (axis cs:3.72,16.4)
    {\tpcifenglish{box}{Kasten}, $n^2$};
  \node[lbl, text=birrengray, anchor=east] at (axis cs:5.50,5.9)
    {\tpcifenglish{oscillator}{Oszillator}, $\tfrac{2n+1}{3}$};
\end{axis}

\end{tikzpicture}
\end{center}
